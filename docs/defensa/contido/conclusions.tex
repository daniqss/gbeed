\section{Conclusións e traballo futuro}

\begin{frame}{Obxectivos cumpridos}
  \small
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{block}{Emulación validada}
        Núcleo completo do \textsc{dmg}-01, validado contra as suites de
        referencia e cos xogos máis vendidos do catálogo.
      \end{block}
      \begin{block}{Núcleo reutilizable}
        Librería \texttt{no\_std}, sen dependencias e independente de calquera
        librería gráfica, con dúas vistas de referencia sobre ela.
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Execución en recursos limitados}
        Velocidade real na familia Raspberry Pi Zero grazas ás optimizacións conseguidas tras o perfilado.
      \end{block}
      \begin{block}{Consola \textsc{diy} documentada}
        Construción de referencia sobre hardware accesible, cunha imaxe de
        sistema reproducible e distribuída baixo \textsc{gpl}v2.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}
