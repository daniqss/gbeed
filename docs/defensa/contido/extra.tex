\begin{frame}[plain,noframenumbering,fragile]{Anexo · O temporizador en tempo constante}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \fonte{Antes: un bucle por ciclo.}
\begin{lstlisting}[language=Rust]
for _ in 0..cycles {
  self.internal_counter =
    self.internal_counter.wrapping_add(1);

  if self.is_timer_enabled() {
    let current = self.get_selected_bit();
    if self.previous_bit && !current {
      self.increment_tima();
    }
    self.previous_bit = current;
  }
}
\end{lstlisting}
    \end{column}
    \begin{column}{0.48\textwidth}
      \fonte{Despois: dous desprazamentos e unha resta.}
\begin{lstlisting}[language=Rust]
let start = prev_counter as u32;
let end = start + cycles as u32;

// cada flanco de baixada
// incrementa TIMA
let edges = (end >> shift)
          - (start >> shift);
\end{lstlisting}
      \vspace{0.3em}
      {\footnotesize A mesma oportunidade apareceu despois na \textsc{apu} e
      resolveuse igual.}
    \end{column}
  \end{columns}
\end{frame}

    \begin{column}{0.44\textwidth}
      {\footnotesize
      Catro canles gobernadas por contadores e un \textit{frame sequencer} a

      do seu ciclo.\\[0.5em]

      volume e panorama e pasa o resultado por un filtro paso alto que elimina
      a compoñente continua dos conversores.}
