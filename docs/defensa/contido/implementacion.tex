\begin{frame}{Execución dunha instrución}
  \vfill
  \begin{itemize}
    \setlength{\itemsep}{1.1em}
    \item Cada instrución é un \texttt{struct} que implementa
    \texttt{Instruction}.

    \item Os operandos das instruccións represéntanse cun tipo xenérico \texttt{Operand} que coñece
    o seu custo en tamaño e tempo de execución.

    \item Os efectos da instrución devolvense nun \texttt{InstructionEffect} que contén o tempo consumido,
    o tamaño total da instrución e as bandeiras afectadas.
  \end{itemize}
  \vfill
\end{frame}

impl<S: Operand> Instruction for And<S> {
  fn exec(&mut self, gb: &mut Dmg) -> InstructionResult {
    gb.cpu.a &= self.src.read(gb);

    Ok(InstructionEffect::new(
      self.info(),
      Flags {
        z: Some(check_zero(gb.cpu.a)),
        n: Some(false),
        h: Some(true),
        c: Some(false),
      },
    ))
  }
  fn info(&self) -> (u8, u8) { (1 + S::READ_CYCLES, 1 + S::LEN) }
  fn disassembly(&self) -> String { format!("and {}", self.src) }
}

\begin{frame}{Os cartuchos amplían o hardware da consola}
  \vfill

    \setlength{\itemsep}{1.4em}

    \item O \textsc{mbc} encárgase de xestionar ese hardware. Existen moitos,

\begin{frame}{Optimización}
  \begin{columns}[c]
    \begin{column}{0.46\textwidth}
      \small
      \begin{itemize}
        \item \destacado{Bucles de debuxado}: reutilizar cálculos entre píxeles
        consecutivos e acceso directo ás memorias da \textsc{ppu}.

        \item \destacado{Complexidade}: o \textit{timer} e a \textsc{apu} son contadores
        periódicos. En vez de iterar un bucle por cada ciclo avanzado pola \textsc{cpu},
        avánzanse cunha suma e derívanse os eventos
        do intervalo por empregando aritmética, baixando a complexidade algorítmica de $O(n)$ a $O(1)$.

        \item \destacado{Bucle quente}: eliminouse as chamadas ao sistema durante bucle quente,
        como as reservas de memoria no \textit{heap} e a impresión de depuración.
      \end{itemize}
    \end{column}
    \begin{column}{0.52\textwidth}
      \begin{center}
        \includegraphics[width=\textwidth]{imaxes/flamegraph_baseline.png}\\[0.2em]
        \fonte{Antes: o debuxado supón o 26,0\,\% do tempo.}\\[0.7em]
        \includegraphics[width=\textwidth]{imaxes/flamegraph_optimized.png}\\[0.2em]
        \fonte{Despois: 9,5\,\%.}
      \end{center}

\end{frame}
